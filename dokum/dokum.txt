#cara roscore
jika local host maka pada ~/.bashrc
export ROS_MASTER_URI=http://localhost:11311
export ROS_HOSTNAME=localhost

jika menggunakan ip maka 
export ROS_MASTER_URI=http://192.168.100.101:11311 //sesuaikan ipnya
export ROS_HOSTNAME=192.168.100.101

kapan menggunakan ip? saat ingin display rviz / mapping pada client pc

pada pc setup di ~/.bashrc atau ~/.zshrc 
# ROS Network Configuration Switcher
function ros_master() {
# Default IPs (modify these if needed)
local MASTER_IP="192.168.150.160"
local CLIENT_IP="192.168.150.177"

# Set mode based on argument
if [ "$1" = "1" ]; then
export ROS_HOSTNAME="localhost"
export ROS_MASTER_URI="http://localhost:11311"
echo "ROS MASTER MODE (localhost)"
elif [ "$1" = "0" ]; then
export ROS_HOSTNAME="$CLIENT_IP"
export ROS_MASTER_URI="http://$MASTER_IP:11311"
echo "ROS CLIENT MODE: $CLIENT_IP → Master @ $MASTER_IP"
else
echo "Usage: ROS_MASTER 1 (master) or ROS_MASTER 0 (client)"
fi
}

kemudian source ~/.bashrc ketika sudah merubah dan save

kemudian jika kondisinya ingin display rviz pada laptop maka pada terminal
ros_master 0

jika ingin mengembalikan di localhost maka 
ros_master 1
